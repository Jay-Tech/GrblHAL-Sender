#Linux Path Config

#Remove config 
rm -r GrblHAL-Sender

#Run from bash
cd /usr/bin/grblhal-sender

#Install linux 
sudo wget https://github.com/Jay-Tech/GrblHAL-Sender/releases/download/v1.0.0/grblhal-sender_1.0.0_arm64.deb
sudo apt install ./grblhal-sender_1.0.0_arm64.deb

#Auto Start Linux 
mkdir -p ~/.config/autostart
cat > ~/.config/autostart/grblhal-sender.desktop << 'EOF'
[Desktop Entry]
Type=Application
Name=GrblHAL Sender
Comment=GrblHAL CNC Sender
Exec=/usr/lib/grblhal-sender/GrbLHALSender.Desktop
Terminal=false
X-GNOME-Autostart-enabled=true
EOF

#Auto Start Linux 2(PI)
